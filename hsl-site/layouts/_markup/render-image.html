{{- /* Bare file name -> /images/
/ (blog: /images/blog//). "folder/x.jpg" -> /images/folder/x.jpg Size via fragment: x.jpg#genis (wide), #yan (floats beside text), #kucuk (small, centred) */ -}} {{- $u := urls.Parse .Destination -}} {{- $src := $u.Path -}} {{- if strings.Contains .Destination "://" -}}{{- $src = printf "%s://%s%s" $u.Scheme $u.Host $u.Path -}}{{- end -}} {{- $size := $u.Fragment -}} {{- $show := true -}} {{- if not (or (strings.HasPrefix $src "/") (strings.Contains $src "://")) -}} {{- if strings.Contains $src "/" -}} {{- $src = printf "/images/%s" $src -}} {{- else -}} {{- $sec := .Page.Section | default "home" -}} {{- if and (eq $sec "blog") .Page.File -}} {{- $post := replaceRE `^\d{4}-\d{2}-\d{2}-` "" .Page.File.ContentBaseName -}} {{- $src = printf "/images/blog/%s/%s" $post $src -}} {{- else -}} {{- $src = printf "/images/%s/%s" $sec $src -}} {{- end -}} {{- end -}} {{- $show = os.FileExists (printf "static%s" $src) -}} {{- $src = $src | relURL -}} {{- end -}} {{- if not $show -}}{{- /* image not uploaded yet: render nothing */ -}} {{- else if or .Title $size -}} {{- else -}} {{- end -}}
